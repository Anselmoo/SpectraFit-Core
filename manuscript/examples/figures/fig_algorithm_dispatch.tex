\documentclass[border=6pt,varwidth=13.4cm]{standalone}

\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage{amsmath}
\usepackage[ruled,vlined]{algorithm2e}

\SetKwInOut{Input}{Input}
\SetKwInOut{Output}{Output}
\newcommand{\solver}[1]{\texttt{#1}}

\begin{document}
\begin{algorithm}[H]
  \DontPrintSemicolon
  \caption{Solver dispatch. The dispatcher checks for a named solver first,
    then---only when the caller passed \texttt{auto}---for separability, and
    otherwise returns the trust-region family. An unrecognised solver name is an
    error, never a silent fallback, so a typo cannot quietly change what was
    measured.}
  \Input{model graph $G$, solver request $s$}
  \Output{a solver instance, or an error}
  \BlankLine
  \eIf{$s$ names a solver}{
    \Return that solver, or \solver{Error} if the name is unknown\;
    \tcp*[h]{never a silent fallback}\;
  }{
    \uIf{$G$ is separable in its linear coefficients,
         \emph{and} every varying nonlinear parameter is unbounded,
         \emph{and} no parameter is tied,
         \emph{and} no node is bound to a dataset index}{
      \Return \solver{VariableProjection}\;
      \tcp*[h]{the linear block is eliminated}\;
    }
    \Else{
      \Return \solver{TrustRegionLevenbergMarquardt}\;
    }
  }
\end{algorithm}
