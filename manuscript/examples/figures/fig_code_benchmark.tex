\begin{lstlisting}
# The release profile is not optional for a timing run: a debug build of
# the extension is one to two orders of magnitude slower, and measures the
# build profile rather than the solver.
uv sync --extra benchmark && uv run maturin develop --release

# --reps and --mc set the depth. The suite phase runs HALF the requested
# repetitions, so 2, 4 and 10 time each case 1, 2 and 5 times; both counts
# are recorded per rung as reps_requested and reps_effective, and the
# figures in Validation are indexed by the effective value -- to reproduce
# the reported depth-50 rung, pass --reps 100.
uv run python -m oracles.cli run --reps 2

# Four-axis regression gate against the pinned baseline.
uv run python -m oracles.cli gate
\end{lstlisting}
